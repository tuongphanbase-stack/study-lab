Consistent Hashing

Servers and keys share one ring instead of a plain hash % N. Add or remove a server and watch how little needs to be remapped, compared to a plain modulo scheme where almost everything would move.

How it works

A simple way to split data across servers is hash(key) % number_of_servers. It works fine — until you add or remove a server. The moment the server count changes, almost every key's target server changes too, forcing a huge amount of data to move around.

Consistent hashing fixes this with a different setup: imagine a circle (a "ring"). Both servers and keys get placed somewhere on that same ring, based on their hash. A key belongs to whichever server is the next one clockwise from it.

Now here's the payoff: add or remove a server, and only the keys sitting between it and the previous server on the ring need to move. Everything else on the ring stays exactly where it is. Compare that to the plain modulo approach, where changing the server count reshuffles almost everything.

Example

Ring positions 0–999. Servers land at 100, 400, and 700. A key that hashes to 250 belongs to the server at 400 — the next one clockwise. If the server at 400 gets removed, only that one key moves (to the server at 700); the key mapped to the server at 100 is untouched.

function addServer(ring, serverName):
    position <- hash(serverName) mod ringSize      // where this server lands on the ring
    ring.insert(position, serverName)

function getServer(ring, key):
    position <- hash(key) mod ringSize             // where this key lands on the ring
    return the first server at or after position, walking clockwise
           (wrap around to the start if none found)   // ring has no "end", it loops
import bisect   # binary search helpers, used to keep positions sorted

class ConsistentHashRing:
    def __init__(self, ring_size=1000):
        self.ring_size = ring_size
        self.positions = []   # sorted server positions on the ring
        self.servers = {}     # position -> server name

    def _hash(self, s):
        return sum((i+1) * ord(c) for i, c in enumerate(s)) % self.ring_size

    def add_server(self, name):
        pos = self._hash(name)                 # where this server lands on the ring
        bisect.insort(self.positions, pos)      # insert while keeping positions sorted
        self.servers[pos] = name

    def get_server(self, key):
        if not self.positions:
            return None
        pos = self._hash(key)                          # where this key lands on the ring
        idx = bisect.bisect_left(self.positions, pos)   # first server position >= pos
        if idx == len(self.positions):
            idx = 0   # nothing found going clockwise: wrap around to the first server
        return self.servers[self.positions[idx]]
class ConsistentHashRing {
  constructor(ringSize = 1000) {
    this.ringSize = ringSize;
    this.servers = []; // { name, pos } entries, kept sorted by pos
  }

  hash(s) {
    let h = 0;
    for (const c of s) h = (h * 31 + c.charCodeAt(0)) % this.ringSize;
    return h;
  }

  addServer(name) {
    const pos = this.hash(name);                        // where this server lands on the ring
    this.servers.push({ name, pos });
    this.servers.sort((a, b) => a.pos - b.pos);           // keep the list sorted by position
  }

  getServer(key) {
    if (this.servers.length === 0) return null;
    const pos = this.hash(key);                          // where this key lands on the ring
    const next = this.servers.find(s => s.pos >= pos);     // first server at or after pos
    return next ? next.name : this.servers[0].name;       // none found: wrap to the first
  }
}
class ConsistentHashRing {
    TreeMap<Integer, String> ring = new TreeMap<>();   // sorted map: position -> server name
    int ringSize;

    ConsistentHashRing(int ringSize) { this.ringSize = ringSize; }

    int hash(String s) {
        int h = 0;
        for (char c : s.toCharArray()) h = (h * 31 + c) % ringSize;
        return h;
    }

    void addServer(String name) {
        ring.put(hash(name), name);   // TreeMap keeps this sorted by position automatically
    }

    String getServer(String key) {
        if (ring.isEmpty()) return null;
        int pos = hash(key);                                 // where this key lands on the ring
        Map.Entry<Integer, String> entry = ring.ceilingEntry(pos); // first position >= pos
        return entry != null ? entry.getValue() : ring.firstEntry().getValue(); // else wrap
    }
}
class ConsistentHashRing {
    map<int, string> ring;   // sorted map: position -> server name
    int ringSize;

    int hashKey(const string& s) {
        int h = 0;
        for (char c : s) h = (h * 31 + c) % ringSize;
        return h;
    }

public:
    ConsistentHashRing(int ringSize) : ringSize(ringSize) {}

    void addServer(const string& name) {
        ring[hashKey(name)] = name;   // std::map keeps this sorted by position automatically
    }

    string getServer(const string& key) {
        if (ring.empty()) return "";
        int pos = hashKey(key);                    // where this key lands on the ring
        auto it = ring.lower_bound(pos);             // first position >= pos
        if (it == ring.end()) it = ring.begin();      // none found: wrap to the first server
        return it->second;
    }
};
