Huffman Coding

Frequent characters get short codes, rare ones get long codes — built by repeatedly merging the two least-frequent nodes into a tree. Watch the compression tree grow, then see the encoded bits.

How it works

Normal text gives every character the same number of bits, no matter how often it shows up. Huffman coding does better: it gives common characters shorter codes, and rare characters longer ones — the same idea Morse code uses, giving the common letter "E" just a single dot.

Building the tree:

  1. Start with one node for each character, weighted by how often it appears.
  2. Find the two nodes with the smallest weight, and merge them under a new parent node.
  3. That new node's weight is the sum of the two it came from.
  4. Keep merging the two smallest nodes until only one is left — that's the root.

Each character's code comes from the path down to it: every left branch adds a 0, every right branch adds a 1. Because every character sits at the very end of a branch (never partway down toward another character), no code is ever the start of another one — which is exactly what lets the encoded bits be read back one character at a time, with no separators needed.

Example

Text "aab": a appears twice, b once. With only two distinct characters, each one just gets a single bit: a = 0, b = 1. Encoding "aab" gives 001 — 3 bits total, versus 24 bits (8 bits × 3 characters) with plain ASCII.

function buildHuffmanTree(text):
    freq <- count of each character in text          // how often each character appears
    queue <- a min-priority-queue of leaf nodes, ordered by frequency

    while size(queue) > 1:
        a <- queue.removeMin()                          // the two least-frequent nodes...
        b <- queue.removeMin()
        parent <- newNode(weight = a.weight + b.weight, left = a, right = b)  // merge them
        queue.insert(parent)                            // the merged node re-enters the queue

    return queue.removeMin()  // the last node left is the root

function assignCodes(node, prefix, codes):
    if node is a leaf:
        codes[node.char] <- prefix                       // path from root to leaf is the code
        return
    assignCodes(node.left, prefix + "0", codes)          // left branch adds a 0
    assignCodes(node.right, prefix + "1", codes)         // right branch adds a 1
import heapq
from collections import Counter

class Node:
    def __init__(self, char, weight, left=None, right=None):
        self.char, self.weight, self.left, self.right = char, weight, left, right
    def __lt__(self, other):
        return self.weight < other.weight   # lets heapq compare nodes by weight

def build_huffman_tree(text):
    freq = Counter(text)                                  # how often each character appears
    heap = [Node(ch, w) for ch, w in freq.items()]          # one leaf per character
    heapq.heapify(heap)                                    # arrange into a min-heap by weight

    while len(heap) > 1:
        a = heapq.heappop(heap)          # the two least-frequent nodes...
        b = heapq.heappop(heap)
        heapq.heappush(heap, Node(None, a.weight + b.weight, a, b))  # merge and reinsert

    return heap[0]  # the last node left is the root

def assign_codes(node, prefix="", codes=None):
    if codes is None:
        codes = {}
    if node.char is not None:                # leaf: path from root to here is its code
        codes[node.char] = prefix or "0"       # guard against a single-character tree
        return codes
    assign_codes(node.left, prefix + "0", codes)   # left branch adds a 0
    assign_codes(node.right, prefix + "1", codes)  # right branch adds a 1
    return codes
class Node {
  constructor(char, weight, left = null, right = null) {
    this.char = char; this.weight = weight; this.left = left; this.right = right;
  }
}

function buildHuffmanTree(text) {
  const freq = {};
  for (const ch of text) freq[ch] = (freq[ch] || 0) + 1;   // how often each character appears

  let nodes = Object.entries(freq).map(([ch, w]) => new Node(ch, w));  // one leaf per char

  while (nodes.length > 1) {
    nodes.sort((a, b) => a.weight - b.weight);   // find the two least-frequent nodes...
    const [a, b] = [nodes.shift(), nodes.shift()];
    nodes.push(new Node(null, a.weight + b.weight, a, b));   // merge and reinsert
  }

  return nodes[0]; // the last node left is the root
}

function assignCodes(node, prefix = "", codes = {}) {
  if (node.char !== null) { codes[node.char] = prefix || "0"; return codes; } // leaf: its code
  assignCodes(node.left, prefix + "0", codes);    // left branch adds a 0
  assignCodes(node.right, prefix + "1", codes);   // right branch adds a 1
  return codes;
}
class Node implements Comparable<Node> {
    Character ch; int weight; Node left, right;
    Node(Character ch, int weight, Node left, Node right) {
        this.ch = ch; this.weight = weight; this.left = left; this.right = right;
    }
    public int compareTo(Node other) { return this.weight - other.weight; }  // order by weight
}

static Node buildHuffmanTree(Map<Character, Integer> freq) {
    PriorityQueue<Node> heap = new PriorityQueue<>();   // min-heap ordered by weight
    for (var entry : freq.entrySet()) {
        heap.add(new Node(entry.getKey(), entry.getValue(), null, null));  // one leaf per char
    }

    while (heap.size() > 1) {
        Node a = heap.poll(), b = heap.poll();     // the two least-frequent nodes...
        heap.add(new Node(null, a.weight + b.weight, a, b));   // merge and reinsert
    }

    return heap.poll(); // the last node left is the root
}

static void assignCodes(Node node, String prefix, Map<Character, String> codes) {
    if (node.ch != null) { codes.put(node.ch, prefix.isEmpty() ? "0" : prefix); return; } // leaf
    assignCodes(node.left, prefix + "0", codes);    // left branch adds a 0
    assignCodes(node.right, prefix + "1", codes);   // right branch adds a 1
}
struct Node {
    char ch; int weight;
    Node *left, *right;
    Node(char c, int w, Node* l=nullptr, Node* r=nullptr) : ch(c), weight(w), left(l), right(r) {}
};

struct Compare {
    bool operator()(Node* a, Node* b) { return a->weight > b->weight; }  // min-heap ordering
};

Node* buildHuffmanTree(unordered_map<char,int>& freq) {
    priority_queue<Node*, vector<Node*>, Compare> heap;
    for (auto& [ch, w] : freq) heap.push(new Node(ch, w));   // one leaf per character

    while (heap.size() > 1) {
        Node* a = heap.top(); heap.pop();   // the two least-frequent nodes...
        Node* b = heap.top(); heap.pop();
        heap.push(new Node('\0', a->weight + b->weight, a, b));  // merge and reinsert
    }

    return heap.top(); // the last node left is the root
}

void assignCodes(Node* node, string prefix, unordered_map<char,string>& codes) {
    if (!node->left && !node->right) { codes[node->ch] = prefix.empty() ? "0" : prefix; return; }
    assignCodes(node->left, prefix + "0", codes);    // left branch adds a 0
    assignCodes(node->right, prefix + "1", codes);   // right branch adds a 1
}
